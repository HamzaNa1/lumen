cmake_minimum_required(VERSION 3.20)
project(LumenVideoHelper LANGUAGES CXX)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol-test protocol-test.cpp)
  target_compile_features(protocol-test PRIVATE cxx_std_20)
  add_test(NAME protocol-boundaries COMMAND protocol-test)
endif()
if(NOT WIN32)
  return()
endif()
add_executable(lumen-video-helper helper.cpp)
set_target_properties(lumen-video-helper PROPERTIES OUTPUT_NAME "Lumen Video Helper")
target_compile_features(lumen-video-helper PRIVATE cxx_std_20)
target_compile_definitions(lumen-video-helper PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(lumen-video-helper PRIVATE vendor)
target_link_libraries(lumen-video-helper PRIVATE d3d11 dxgi)
if(MSVC)
  set_property(TARGET lumen-video-helper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(lumen-video-helper PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(lumen-video-helper PRIVATE -Wall -Wextra -Werror)
  target_link_options(lumen-video-helper PRIVATE -static -static-libgcc -static-libstdc++)
endif()
